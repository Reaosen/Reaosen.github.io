🐧 常用 Linux 命令速查表

所有示例均基于 Bash，适用于绝大多数 GNU/Linux 发行版。

tail -f /var/log/app.log        # 实时跟踪日志（最常用）
tail -n 100 /var/log/app.log    # 查看最后100行
head -n 20 /var/log/app.log     # 查看前20行
grep "ERROR" /var/log/app.log
grep -i "timeout" /var/log/app.log      # 忽略大小写
grep -A 5 -B 5 "Exception" app.log      # 显示匹配前后5行上下文
zgrep "error" app.log.1.gz              # 搜索压缩日志（.gz）

16. 性能排查一图流（80% 场景够用）

# 1. 负载高
uptime
# 2. 谁占 CPU
ps -eo pid,ppid,%cpu,%mem,cmd --sort=-%cpu | head
# 3. 谁占内存
ps -eo pid,ppid,%cpu,%mem,cmd --sort=-%mem | head
# 4. 磁盘瓶颈
iostat -x 1  # 安装：sysstat
# 5. 网络连接数
ss -s
# 6. 最热文件
lsof | awk '{print $NF}' | sort | uniq -c | sort -nr | head -20
